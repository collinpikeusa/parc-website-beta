# parcradio.net

User-agent: *
Allow: /

# The VE pages are deliberately NOT disallowed here.
#
# They carry <meta name="robots" content="noindex, nofollow">, and a crawler has
# to be able to FETCH a page to see that tag. Disallowing them in robots.txt
# would block the crawl, the noindex would never be read, and URLs already known
# to Google could sit in the index indefinitely with no way to remove them.
# The pages are encrypted, so there is nothing to read even if fetched.

# Encrypted blobs — no value in crawling, no content to index.
Disallow: /ve/files/

Sitemap: https://parcradio.net/sitemap.xml
